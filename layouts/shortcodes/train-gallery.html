{{- /* Photos live in assets/images/trains//, not in the page bundle: bundle files without a language suffix belong to the default language (en), so a page that exists only in ru gets none of them. photos.yaml there holds the card cover and named photo groups ("galleries"); each group lists files in display order with their credits. alt and copyright are either plain text or a language map ({ru: …, en: …, ka: …}, see partials/localized.html). Usage: {{< train-gallery >}} renders the "main" group, {{< train-gallery "business" >}} another one in the compact frame. Layout: a scroll-snap track of slides (swipe on touch, arrows/thumbnails on desktop); each slide is a fixed-height frame with the photo fitted inside over a blurred copy of itself, so photos of any orientation never shift the text below. Fullscreen reuses the same track (assets/js/train-gallery.js). */ -}} {{- $page := .Page -}} {{- $group := .Get 0 | default "main" -}} {{- $dir := printf "images/trains/%s" $page.File.ContentBaseName -}} {{- with resources.Get (printf "%s/photos.yaml" $dir) -}} {{- $photos := slice -}} {{- with index ((transform.Unmarshal .).galleries | default dict) $group -}} {{- range . -}} {{- $meta := . -}} {{- with resources.Get (printf "%s/%s" $dir $meta.src) -}} {{- $photos = $photos | append (dict "img" . "alt" (partial "localized.html" $meta.alt) "copyright" (partial "localized.html" $meta.copyright) "copyright_link" $meta.copyright_link ) -}} {{- else -}} {{- warnf "train-gallery: %s is listed in assets/%s/photos.yaml but not found" $meta.src $dir -}} {{- end -}} {{- end -}} {{- else -}} {{- warnf "train-gallery: no group %q in assets/%s/photos.yaml" $group $dir -}} {{- end -}} {{- $count := len $photos -}} {{- if $count -}}
{{- range $i, $p := $photos -}} {{- $img := $p.img -}} {{- $srcset := slice -}} {{- range $w := slice 640 1024 1600 -}} {{- if lt $w $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q82" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- end -}} {{- end -}} {{- if le $img.Width 1600 -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- end -}} {{- $src := $img -}} {{- if gt $img.Width 1024 }}{{ $src = $img.Resize "1024x webp q82" }}{{ end -}} {{- $blur := $img.Resize "48x webp q50" -}} {{- end -}}
{{- if gt $count 1 }} 1 / {{ $count }} {{- end }}
{{- if gt $count 1 }}
{{- range $i, $p := $photos -}} {{- $thumb := $p.img.Resize "x112 webp q75" -}} {{- end -}}
{{- end }}
